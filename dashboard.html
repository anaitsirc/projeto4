function parseVotos(valor) {
    if (valor === "-" || valor === "c.r." || valor === undefined || valor === null) {
        return 0;
    }
    if (typeof valor === 'string') {
        return parseInt(valor.replace(/\s/g, '')) || 0;
    }
    return Number(valor) || 0;
}

// Função para obter resultados de um distrito
function getDistrictResults(districtName) {
    const results = [];
    const partidos = getPartidos();
    
    for (const partido of partidos) {
        const votos = parseVotos(electionData[partido]?.[districtName]);
        if (votos > 0) {
            results.push({ 
                partido, 
                votos,
                percentagem: ((votos / (electionData.Votantes[districtName] || 1)) * 100).toFixed(2)
            });
        }
    }
    
    return results.sort((a, b) => b.votos - a.votos);
}

// Função para obter o partido vencedor
function getWinner(districtName) {
    const results = getDistrictResults(districtName);
    return results.length > 0 ? results[0].partido : null;
}

// Função para obter a cor do vencedor
function getWinnerColor(districtName) {
    const winner = getWinner(districtName);
    const colors = getPartyColors();
    return winner ? (colors[winner] || "#b3afaf") : "#b3afaf";
}

// Função para calcular intensidade
function calcularIntensidade(partido, distrito) {
    const votosPartido = parseVotos(electionData[partido]?.[distrito]);
    const votosDistrito = electionData.Votantes[distrito] || 1;
    
    if (votosPartido === 0 || votosDistrito === 0) return 0.1;
    return Math.min(0.9, Math.max(0.1, votosPartido / votosDistrito));
}

// Cálculo de métricas nacionais
totalInscritos = electionData.Inscritos.Total
totalVotantes = electionData.Votantes.Total
taxaAbstencao = ((totalInscritos - totalVotantes) / totalInscritos * 100).toFixed(2)
taxaParticipacao = (totalVotantes / totalInscritos * 100).toFixed(2)

// Encontrar vencedor nacional
function getNationalWinner() {
    const resultados = getDistrictResults("Total");
    return resultados.length > 0 ? resultados[0] : null;
}

vencedorNacional = getNationalWinner()

mapDiv = html`<div id="map-container" style="height: 500px; border-radius: 8px; border: 1px solid #ddd;"></div>`

// Inicializar mapa quando a célula for executada
{
    // Aguardar Leaflet carregar
    if (typeof L !== 'undefined') {
        const map = L.map('map-container').setView([39.5, -8.0], 6);
        
        // Adicionar camada base
        L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
            attribution: '© OpenStreetMap contributors'
        }).addTo(map);
        
        // Função para obter estilo do polígono
        function getStyle(feature) {
            const distrito = feature.properties.Distrito;
            const isSelected = distrito === mutable selectedDistrict;
            
            let fillColor, fillOpacity;
            
            if (selectedMapMode === 'VENCEDOR') {
                fillColor = getWinnerColor(distrito);
                fillOpacity = 0.7;
            } else {
                const intensidade = calcularIntensidade(selectedParty, distrito);
                fillColor = getPartyColors()[selectedParty] || "#b3afaf";
                fillOpacity = Math.max(0.2, intensidade);
            }
            
            return {
                fillColor: fillColor,
                fillOpacity: fillOpacity,
                color: isSelected ? '#000000' : '#333333',
                weight: isSelected ? 3 : 1,
                opacity: 0.8
            };
        }
        
        // Adicionar camada GeoJSON
        const geojsonLayer = L.geoJSON(districtData, {
            style: getStyle,
            onEachFeature: function(feature, layer) {
                const nome = feature.properties.Distrito;
                const vencedor = getWinner(nome);
                const votantes = electionData.Votantes[nome] || 0;
                const inscritos = electionData.Inscritos[nome] || 1;
                const abstencao = (((inscritos - votantes) / inscritos) * 100).toFixed(1);
                
                // Tooltip
                let tooltipContent = `
                    <strong>${nome}</strong><br/>
                    Vencedor: ${vencedor || 'N/A'}<br/>
                    Votantes: ${votantes.toLocaleString('pt-PT')}<br/>
                    Abstenção: ${abstencao}%
                `;
                
                if (selectedMapMode === 'INTENSIDADE') {
                    const votosPartido = parseVotos(electionData[selectedParty]?.[nome]);
                    const percentagem = ((votosPartido / votantes) * 100).toFixed(1);
                    tooltipContent += `<br/>${selectedParty}: ${votosPartido.toLocaleString('pt-PT')} (${percentagem}%)`;
                }
                
                layer.bindTooltip(tooltipContent);
                
                // Eventos
                layer.on({
                    click: function() {
                        mutable selectedDistrict = nome;
                    },
                    mouseover: function(e) {
                        layer.setStyle({ weight: 3, color: '#0066cc' });
                    },
                    mouseout: function(e) {
                        const style = getStyle(feature);
                        layer.setStyle(style);
                    }
                });
            }
        }).addTo(map);
        
        // Ajustar mapa aos limites
        map.fitBounds(geojsonLayer.getBounds());
    }
}

function prepareSunburstData() {
    const root = {
        name: "Portugal",
        children: []
    };
    
    getPartidos().forEach(partido => {
        const partidoNode = {
            name: partido,
            children: []
        };
        
        for (const distrito in electionData[partido]) {
            if (distrito !== "Total" && distrito !== "Europa" && distrito !== "Fora da Europa") {
                const votos = parseVotos(electionData[partido][distrito]);
                if (votos > 0) {
                    partidoNode.children.push({
                        name: distrito,
                        value: votos
                    });
                }
            }
        }
        
        partidoNode.value = partidoNode.children.reduce((sum, child) => sum + child.value, 0);
        
        if (partidoNode.value > 0) {
            root.children.push(partidoNode);
        }
    });
    
    root.value = root.children.reduce((sum, child) => sum + child.value, 0);
    
    return root;
}

sunburstData = prepareSunburstData()

// CÉLULA 8: Gráfico Sunburst
// | label: sunburst-chart
// | echo: false

{
    // Verificar se D3 está carregado
    if (typeof d3 === 'undefined') {
        return html`<div style="color: red; padding: 20px; border: 2px solid red;">
            <h3>❌ D3.js não carregado</h3>
            <p>Adicione no cabeçalho: <code>include-after-body: - &lt;script src="https://unpkg.com/d3@7"&gt;&lt;/script&gt;</code></p>
        </div>`;
    }
    
    const width = 600;
    const height = 600;
    const radius = Math.min(width, height) / 2;
    
    const svg = d3.create("svg")
        .attr("width", width)
        .attr("height", height)
        .attr("viewBox", [-width / 2, -height / 2, width, height])
        .attr("style", "max-width: 100%; height: auto;");
    
    const root = d3.hierarchy(sunburstData)
        .sum(d => d.value)
        .sort((a, b) => b.value - a.value);
    
    const partition = d3.partition()
        .size([2 * Math.PI, radius]);
    
    const arc = d3.arc()
        .startAngle(d => d.x0)
        .endAngle(d => d.x1)
        .innerRadius(d => d.y0)
        .outerRadius(d => d.y1);
    
    partition(root);
    
    const partyColors = getPartyColors();
    const colorScale = d3.scaleOrdinal()
        .domain(getPartidos())
        .range(getPartidos().map(p => partyColors[p] || "#cccccc"));
    
    // Desenhar arcos
    svg.selectAll("path")
        .data(root.descendants().filter(d => d.depth))
        .join("path")
        .attr("d", arc)
        .style("fill", d => {
            if (d.depth === 1) return colorScale(d.data.name);
            if (d.depth === 2) {
                const baseColor = colorScale(d.parent.data.name);
                return d3.color(baseColor).brighter(0.5);
            }
            return "#eee";
        })
        .style("opacity", 0.8)
        .style("stroke", "#fff")
        .style("stroke-width", 1)
        .on("mouseover", function(event, d) {
            d3.select(this).style("opacity", 1);
        })
        .on("mouseout", function() {
            d3.select(this).style("opacity", 0.8);
        });
    
    // Adicionar texto
    svg.selectAll("text")
        .data(root.descendants().filter(d => d.depth && (d.y1 - d.y0) * (d.x1 - d.x0) > 0.03))
        .join("text")
        .attr("transform", function(d) {
            const x = (d.x0 + d.x1) / 2 * 180 / Math.PI;
            const y = (d.y0 + d.y1) / 2;
            return `rotate(${x - 90}) translate(${y},0) rotate(${x < 180 ? 0 : 180})`;
        })
        .attr("dy", "0.35em")
        .attr("text-anchor", "middle")
        .attr("font-size", d => Math.min(12, (d.y1 - d.y0) * 0.8))
        .text(d => d.data.name)
        .style("fill", "#333")
        .style("pointer-events", "none");
    
    return html`
    <div style="text-align: center;">
        <h3>☀️ Visualização Hierárquica dos Resultados</h3>
        <p style="margin-bottom: 15px; color: #666;">
            Distribuição de votos por partido e distrito
        </p>
        ${svg.node()}
        <div style="margin-top: 15px; font-size: 12px; color: #888;">
            <strong>Legenda:</strong> Círculo interno = Partidos, Círculo externo = Distritos
        </div>
    </div>
    `;
