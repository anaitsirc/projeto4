function parseVotos(valor) {
    if (valor === "-" || valor === "c.r." || valor === undefined || valor === null) return 0;
    if (typeof valor === 'string') return parseInt(valor.replace(/\s/g, '')) || 0;
    return Number(valor) || 0;
}

function getDistrictResults(districtName) {
    const results = [];
    const partidos = getPartidos();
    
    for (const partido of partidos) {
        const votos = parseVotos(electionData[partido]?.[districtName]);
        if (votos > 0) {
            results.push({ 
                partido, 
                votos,
                percentagem: ((votos / (electionData.Votantes[districtName] || 1)) * 100).toFixed(2)
            });
        }
    }
    
    return results.sort((a, b) => b.votos - a.votos);
}

function getWinner(districtName) {
    const results = getDistrictResults(districtName);
    return results.length > 0 ? results[0].partido : null;
}

function getWinnerColor(districtName) {
    const winner = getWinner(districtName);
    const colors = getPartyColors();
    return winner ? (colors[winner] || "#b3afaf") : "#b3afaf";
}

function calcularIntensidade(partido, distrito) {
    const votosPartido = parseVotos(electionData[partido]?.[distrito]);
    const votosDistrito = electionData.Votantes[distrito] || 1;
    
    if (votosPartido === 0 || votosDistrito === 0) return 0.1;
    return Math.min(0.9, Math.max(0.1, votosPartido / votosDistrito));
}

// Funções para cálculo de métricas de abstenção
function getAbstentionByDistrict(districtName) {
    const inscritos = electionData.Inscritos[districtName] || 0;
    const votantes = electionData.Votantes[districtName] || 0;
    return ((inscritos - votantes) / inscritos * 100).toFixed(2);
}

function getHighestAbstentionDistrict() {
    let maxAbstention = 0;
    let district = "";
    let details = {};
    
    for (const distrito in electionData.Inscritos) {
        if (distrito !== "Total") {
            const abstencao = parseFloat(getAbstentionByDistrict(distrito));
            if (abstencao > maxAbstention) {
                maxAbstention = abstencao;
                district = distrito;
                details = {
                    inscritos: electionData.Inscritos[distrito],
                    votantes: electionData.Votantes[distrito]
                };
            }
        }
    }
    return { district, abstencao: maxAbstention, ...details };
}

function getLowestAbstentionDistrict() {
    let minAbstention = 100;
    let district = "";
    let details = {};
    
    for (const distrito in electionData.Inscritos) {
        if (distrito !== "Total") {
            const abstencao = parseFloat(getAbstentionByDistrict(distrito));
            if (abstencao < minAbstention) {
                minAbstention = abstencao;
                district = distrito;
                details = {
                    inscritos: electionData.Inscritos[distrito],
                    votantes: electionData.Votantes[distrito]
                };
            }
        }
    }
    return { district, abstencao: minAbstention, ...details };
}

// Função para obter votos do partido selecionado por distrito
function getPartyVotesByDistrict(partido) {
    const resultados = [];
    
    for (const distrito in electionData.Inscritos) {
        if (distrito !== "Total" && distrito !== "undefined") {
            const votos = parseVotos(electionData[partido]?.[distrito]);
            if (votos > 0) {
                const votantes = electionData.Votantes[distrito] || 1;
                const percentagem = (votos / votantes * 100).toFixed(2);
                resultados.push({
                    distrito: distrito,
                    votos: votos,
                    percentagem: percentagem
                });
            }
        }
    }
    
    return resultados.sort((a, b) => b.votos - a.votos);
}

function getStyleMainMap(feature) {
    const distrito = feature.properties.Distrito;
    const isSelected = distrito === mutable selectedDistrict;
    
    let fillColor, fillOpacity;
    
    if (selectedParty === 'Todos' || selectedMapMode === 'VENCEDOR') {
        fillColor = getWinnerColor(distrito);
        fillOpacity = 0.7;
    } else {
        const intensidade = calcularIntensidade(selectedParty, distrito);
        fillColor = getPartyColors()[selectedParty] || "#b3afaf";
        fillOpacity = Math.max(0.2, intensidade);
    }
    
    return {
        fillColor: fillColor,
        fillOpacity: fillOpacity,
        color: isSelected ? '#000000' : '#333333',
        weight: isSelected ? 3 : 1,
        opacity: 0.8
    };
}

function getStylePartyMap(feature) {
    const distrito = feature.properties.Distrito;
    const isSelected = distrito === mutable selectedDistrict;
    
    let fillColor, fillOpacity;
    
    if (selectedParty === 'Todos') {
        // Se nenhum partido está selecionado, mostrar mapa em escala de cinza
        fillColor = "#b3afaf";

        // Usar a cor do partido selecionado

        // Calcular intensidade baseada na percentagem de votos
        const votosPartido = parseVotos(electionData[selectedParty]?.[distrito]);
        const votantes = electionData.Votantes[distrito] || 1;
        const percentagem = votosPartido / votantes;
        
        // Definir opacidade baseada na intensidade (mínimo 0.2, máximo 0.9)
        fillOpacity = Math.max(0.2, Math.min(0.9, percentagem * 2));
    }
    
    return {
        fillColor: fillColor,
        fillOpacity: fillOpacity,
        color: isSelected ? '#000000' : '#333333',
        weight: isSelected ? 3 : 1,
        opacity: 0.8
    };
}

function prepareSunburstData(electionData, getPartidos, getDistrictResults) {
    const partidos = getPartidos().filter(p => p !== 'Total');
    const districtNames = Object.keys(electionData.Inscritos).filter(d => d !== 'Total');
    
    // Obter os resultados totais por partido para usar como 'value' no nível Partido
    const resultadosNacionais = getDistrictResults("Total");
    const votosPorPartidoNacional = resultadosNacionais.reduce((acc, curr) => {
        acc[curr.partido] = curr.votos;
        return acc;
    }, {});

    const partyNodes = partidos.map(partido => {
        const districtNodes = districtNames.map(distrito => {
            const votos = parseVotos(electionData[partido]?.[distrito]);
            // Apenas incluir distritos onde o partido teve votos
            if (votos > 0) {
                return {
                    name: distrito,
                    value: votos
                };
            }
            return null;
        }).filter(d => d !== null);

        // Se o partido tem votos no total (para evitar nós de partidos vazios)
        if (votosPorPartidoNacional[partido] > 0) {
             return {
                name: partido,
                // Usamos o total de votos do partido para ordenar no sunburst e definir o tamanho do setor principal
                value: votosPorPartidoNacional[partido], 
                children: districtNodes.sort((a, b) => b.value - a.value) // Ordenar distritos por votos
            };
        }
        return null;
    }).filter(d => d !== null);

    return {
        name: "Portugal",
        children: partyNodes.sort((a, b) => b.value - a.value) // Ordena Partidos
    };
}

// 2. PREPARAR OS DADOS
sunburstData = prepareSunburstData(electionData, getPartidos, getDistrictResults)

// 3. IMPLEMENTAÇÃO DO GRÁFICO SUNBURST (Adaptado do D3)
chart = {
  // Configurações e Dimensões
  const width = 928;
  const height = width;
  const radius = width / 6;
  const colors = getPartyColors();
  
  // Criar a escala de cores, usando as cores dos partidos
  const color = d3.scaleOrdinal()
      .domain(sunburstData.children.map(d => d.name))
      .range(sunburstData.children.map(d => colors[d.name] || "#b3afaf"));

  // Computar o layout
  const hierarchy = d3.hierarchy(sunburstData)
      .sum(d => d.value)
      .sort((a, b) => b.value - a.value); // Ordenar por valor
  const root = d3.partition()
      .size([2 * Math.PI, hierarchy.height + 1])
    (hierarchy);
  root.each(d => d.current = d);

  // Criar o gerador de arco (arc generator)
  const arc = d3.arc()
      .startAngle(d => d.x0)
      .endAngle(d => d.x1)
      .innerRadius(d => d.y0 * radius)
      // ** A CORREÇÃO: Usar D.Y1 * RADIUS sem subtrair nada **
      .outerRadius(d => d.y1 * radius) 
      .cornerRadius(0); // Garantir que não há arredondamento angular

  // Criar o contentor SVG
  const svg = d3.create("svg")
      .attr("viewBox", [-width / 2, -height / 2, width, width])
      .style("font", "10px sans-serif");

  // Adicionar os arcos
  const path = svg.append("g")
    .selectAll("path")
    .data(root.descendants().slice(1))
    .join("path")
      // Usar a escala de cores baseada no partido pai (depth 1)
      .attr("fill", d => { while (d.depth > 1) d = d.parent; return color(d.data.name); }) 
      .attr("fill-opacity", d => arcVisible(d.current) ? (d.children ? 0.8 : 0.6) : 0) 
      .attr("pointer-events", d => arcVisible(d.current) ? "auto" : "none")
      .attr("d", d => arc(d.current));

  // Torná-los clicáveis se tiverem filhos.
  path.filter(d => d.children)
      .style("cursor", "pointer")
      .on("click", clicked);

  const format = d3.format(",.0f"); // Formatação para Votos (inteiro)
  path.append("title")
      .text(d => `${d.ancestors().map(d => d.data.name).reverse().join(" -> ")}\n${format(d.value)} votos`); 

  const label = svg.append("g")
      .attr("pointer-events", "none")
      .attr("text-anchor", "middle")
      .style("user-select", "none")
    .selectAll("text")
    .data(root.descendants().slice(1))
    .join("text")
      .attr("dy", "0.35em")
      .attr("fill-opacity", d => +labelVisible(d.current))
      .attr("transform", d => labelTransform(d.current))
      .text(d => d.data.name);

  const parent = svg.append("circle")
      .datum(root)
      .attr("r", radius)
      .attr("fill", "none")
      .attr("pointer-events", "all")
      .on("click", clicked);

  // Lidar com o zoom ao clicar.
  function clicked(event, p) {
    parent.datum(p.parent || root);

    root.each(d => d.target = {
      x0: Math.max(0, Math.min(1, (d.x0 - p.x0) / (p.x1 - p.x0))) * 2 * Math.PI,
      x1: Math.max(0, Math.min(1, (d.x1 - p.x0) / (p.x1 - p.x0))) * 2 * Math.PI,
      y0: Math.max(0, d.y0 - p.depth),
      y1: Math.max(0, d.y1 - p.depth)
    });

    const t = svg.transition().duration(event.altKey ? 7500 : 750);

    path.transition(t)
        .tween("data", d => {
          const i = d3.interpolate(d.current, d.target);
          return t => d.current = i(t);
        })
      .filter(function(d) {
        return +this.getAttribute("fill-opacity") || arcVisible(d.target);
      })
        .attr("fill-opacity", d => arcVisible(d.target) ? (d.children ? 0.8 : 0.6) : 0)
        .attr("pointer-events", d => arcVisible(d.target) ? "auto" : "none")  
        .attrTween("d", d => () => arc(d.current));

    label.filter(function(d) {
        return +this.getAttribute("fill-opacity") || labelVisible(d.target);
      }).transition(t)
        .attr("fill-opacity", d => +labelVisible(d.target))
        .attrTween("transform", d => () => labelTransform(d.current));
  }
  
  function arcVisible(d) {
    // Definimos 3 como profundidade máxima (Portugal=0, Partidos=1, Distritos=2)
    return d.y1 <= 3 && d.y0 >= 1 && d.x1 > d.x0; 
  }

  function labelVisible(d) {
    // Aumentar o limite para mostrar menos rótulos pequenos e evitar a sobreposição
    return d.y1 <= 3 && d.y0 >= 1 && (d.y1 - d.y0) * (d.x1 - d.x0) > 0.05; 
  }

  function labelTransform(d) {
    const x = (d.x0 + d.x1) / 2 * 180 / Math.PI;
    const y = (d.y0 + d.y1) / 2 * radius;
    return `rotate(${x - 90}) translate(${y},0) rotate(${x < 180 ? 0 : 180})`;
  }

  return html`<div style="text-align: center; margin: 20px 0;">
    ${svg.node()}
    <p style="margin-top: 10px; font-size: 0.9em; color: #666;">
        Clique nos anéis exteriores (Partidos) para fazer zoom. Clique no centro para recuar.
    </p>
</div>`;
}
