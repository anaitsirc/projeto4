Resultados Eleitorais Legislativas 2025

Introdução

Este dashboard interativo permite explorar os resultados das eleições legislativas portuguesas de 2025 através de mapas, gráficos e tabelas interativas.

```{ojs}
// | output: false
// CARREGAR TODOS OS DADOS DE UMA VEZ
const geojsonData = FileAttachment("data/ContinenteDistritos.geojson").json();
const electionData = FileAttachment("data/legislativas.json").json();

// Funções auxiliares
function cleanNumericValue(value) {
  if (value === "-" || value === "" || value === "c.r." || value === null) return 0;
  if (typeof value === 'string') return parseInt(value.replace(/\./g, '')) || 0;
  return value;
}

function getPartyColors() {
  return {
    "PS": "#ff0000", "PPD/PSD.CDS-PP": "#ff9900", "CH": "#0000ff",
    "IL": "#00ffff", "B.E.": "#ff00ff", "PCP-PEV": "#008000",
    "L": "#800080", "PAN": "#00ff00"
  };
}

function getDistrictResults(districtName) {
  const results = [];
  for (const [party, districts] of Object.entries(electionData)) {
    if (party !== "Inscritos" && party !== "Votantes") {
      const votes = cleanNumericValue(districts[districtName]);
      if (votes > 0) results.push({ partido: party, votos: votes });
    }
  }
  return results.sort((a, b) => b.votos - a.votos);
}

function getWinnerParty(districtName) {
  const results = getDistrictResults(districtName);
  return results.length > 0 ? results[0].partido : null;
}

function calculateIntensity(party, district) {
  const partyVotes = cleanNumericValue(electionData[party]?.[district] || 0);
  const totalVotes = cleanNumericValue(electionData["Votantes"]?.[district] || 1);
  return partyVotes / totalVotes;
}

// Lista de partidos
const partyList = Object.keys(electionData).filter(p => p !== "Inscritos" && p !== "Votantes");
```

```{ojs}
// | output: false
const totalInscritos = electionData.Inscritos.Total;
const totalVotantes = electionData.Votantes.Total;
const abstencaoNacional = ((totalInscritos - totalVotantes) / totalInscritos * 100).toFixed(2);

function findMinAbstention() {
  let minRate = 100, minDistrict = "";
  for (const [district, inscritos] of Object.entries(electionData.Inscritos)) {
    if (district !== "Total" && electionData.Votantes[district]) {
      const rate = 100 - (electionData.Votantes[district] / inscritos * 100);
      if (rate < minRate) { minRate = rate; minDistrict = district; }
    }
  }
  return { district: minDistrict, rate: minRate.toFixed(2) };
}

function findNationalWinner() {
  let maxVotes = 0, winner = "";
  for (const [party, data] of Object.entries(electionData)) {
    if (party !== "Inscritos" && party !== "Votantes" && data.Total) {
      const votes = cleanNumericValue(data.Total);
      if (votes > maxVotes) { maxVotes = votes; winner = party; }
    }
  }
  return { party: winner, votes: maxVotes };
}

const minAbstention = findMinAbstention();
const nationalWinner = findNationalWinner();
```

Visualização Hierárquica Sunburst

```{ojs}
// | output: false
function prepareSunburstData() {
  const root = { name: "Portugal", children: [] };
  
  // Pegar apenas os 8 principais partidos para simplificar
  const topParties = partyList.slice(0, 8);
  
  topParties.forEach(party => {
    const partyNode = { name: party, children: [] };
    let totalPartyVotes = 0;
    
    // Somar votos de todos os distritos
    for (const [district, votes] of Object.entries(electionData[party])) {
      if (district !== "Total") {
        const numericVotes = cleanNumericValue(votes);
        if (numericVotes > 0) {
          totalPartyVotes += numericVotes;
        }
      }
    }
    
    if (totalPartyVotes > 0) {
      partyNode.value = totalPartyVotes;
      root.children.push(partyNode);
    }
  });
  
  return root;
}

const sunburstData = prepareSunburstData();
```
